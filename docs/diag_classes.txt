    %% ============ UTILISATEURS & RÔLES ============
    class Utilisateur {
        <<abstract>>
        -UUID id
        -UUID etablissementId
        -String email, motDePasse
        -String nom, prenom, telephone
        -Role role
        -Boolean actif
        -DateTime createdAt
        +seConnecter()
        +modifierProfil()
    }

    class Directeur {
        +configurerEtablissement()
        +gererDroitsAcces()
        +validerEvenement()
    }

    class Secretaire {
        +publierAnnonce()
        +gererDossierEleve()
        +gererReinscription()
    }

    class Comptable {
        +validerPaiement()
        +genererFacture()
        +envoyerFactureParEmail()
    }

    class Professeur {
        -String specialite, matricule
        -List~Pointage~ historiquePointage
        +publierCours()
        +saisirNotes()
        +pointer()
    }

    class Eleve {
        -String matricule
        -Date dateNaissance
        -String lieuNaissance
        -Sexe sexe
        -StatutInscription statut
        +consulterCours()
        +consulterBulletins()
        +demanderReinscription()
    }

    class Parent {
        -String profession
        +consulterResultats()
        +effectuerPaiement()
    }

    class Pointage {
        -DateTime date
        -Time heureArrivee, heureDepart
        -StatutPointage statut
    }

    Utilisateur <|-- Directeur
    Utilisateur <|-- Secretaire
    Utilisateur <|-- Comptable
    Utilisateur <|-- Professeur
    Utilisateur <|-- Eleve
    Utilisateur <|-- Parent
    Professeur "1" --> "*" Pointage : enregistre
    Eleve "*" --> "1" Parent : tuteur

        -String referenceTransaction
        -StatutPaiement statut

    }

    class Reinscription {
        -UUID id
        -String anneeScolaire
        -DateTime dateDemande
        -StatutReinscription statut
        +confirmer()
        +refuser()
    }

    Eleve "1" --> "*" Transaction : effectue

    Reinscription "1" --> "0..1" Transaction : inclut
